export interface Reaction { emoji: string; count: number; on?: boolean; }
export interface ChatBubbleProps {
  text: React.ReactNode;
  /** Your own message: right-aligned, gold-soft fill + gold border, tail bottom-right. */
  mine?: boolean;
  name?: string;
  time?: string;
  reactions?: Reaction[];
  onTap?: () => void;
  onReact?: (emoji: string) => void;
  /** Ease in from its own side (new arrival). */
  fresh?: boolean;
}
export function ChatBubble(props: ChatBubbleProps): JSX.Element;
